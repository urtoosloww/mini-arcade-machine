# ILI9341 320x240 landscape, 16bpp (RGB565)
# For panel-mipi-dbi / mipi-dbi-spi overlay
# Build with:  ./mipi-dbi-cmd ili9341.bin ili9341.txt

command 0x01                              # SWRESET
delay 150

command 0xEF 0x03 0x80 0x02               # undocumented, in all vendor inits
command 0xCF 0x00 0xC1 0x30               # Power control B
command 0xED 0x64 0x03 0x12 0x81          # Power on sequence control
command 0xE8 0x85 0x00 0x78               # Driver timing control A
command 0xCB 0x39 0x2C 0x00 0x34 0x02     # Power control A
command 0xF7 0x20                         # Pump ratio control
command 0xEA 0x00 0x00                    # Driver timing control B

command 0xC0 0x23                         # PWCTR1  VRH[5:0]
command 0xC1 0x10                         # PWCTR2  SAP[2:0] BT[3:0]
command 0xC5 0x3E 0x28                    # VMCTR1  contrast
command 0xC7 0x86                         # VMCTR2

command 0x36 0x28                         # MADCTL: MV|BGR -> landscape
command 0x3A 0x55                         # COLMOD: 16 bits/pixel
command 0xB1 0x00 0x18                    # FRMCTR1 ~70Hz
command 0xB6 0x08 0x82 0x27               # DISCTRL

command 0xF2 0x00                         # 3Gamma off
command 0x26 0x01                         # Gamma curve 1
command 0xE0 0x0F 0x31 0x2B 0x0C 0x0E 0x08 0x4E 0xF1 0x37 0x07 0x10 0x03 0x0E 0x09 0x00
command 0xE1 0x00 0x0E 0x14 0x03 0x11 0x07 0x31 0xC1 0x48 0x08 0x0F 0x0C 0x31 0x36 0x0F

command 0x11                              # SLPOUT
delay 150
command 0x29                              # DISPON
delay 150
